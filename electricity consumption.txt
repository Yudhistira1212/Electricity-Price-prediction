!pip install -q streamlit tensorflow scikit-learn pandas numpy

app_code = r'''
import numpy as np
import pandas as pd
import streamlit as st

from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout, Input
from tensorflow.keras.callbacks import EarlyStopping

st.set_page_config(page_title="WattWise", page_icon="⚡", layout="wide")


@st.cache_data
def create_dataset():
    np.random.seed(42)
    dates = pd.date_range("2023-01-01", "2025-12-31", freq="D")
    day = np.arange(len(dates))
    weekly = 15 * np.sin(2 * np.pi * day / 7)
    yearly = 12 * np.sin(2 * np.pi * day / 365)
    noise = np.random.normal(0, 2, len(dates))
    consumption = 120 + weekly + yearly + noise
    price = 8 + 0.03 * (consumption - 120) + np.random.normal(0, 0.15, len(dates))
    price = np.maximum(price, 5)
    return pd.DataFrame({
        "Date": dates,
        "Consumption": consumption,
        "Price_per_kWh": price,
    })


@st.cache_resource(show_spinner=False)
def train_model():
    df = create_dataset()
    scaler = MinMaxScaler()
    scaled = scaler.fit_transform(df[["Consumption"]]).flatten()

    LOOKBACK, FORECAST_DAYS = 30, 30
    X, y = [], []
    for i in range(len(scaled) - LOOKBACK - FORECAST_DAYS + 1):
        X.append(scaled[i:i + LOOKBACK])
        y.append(scaled[i + LOOKBACK:i + LOOKBACK + FORECAST_DAYS])
    X = np.array(X).reshape(-1, LOOKBACK, 1)
    y = np.array(y)
    split = int(len(X) * 0.80)

    model = Sequential([
        Input(shape=(LOOKBACK, 1)),
        LSTM(64, return_sequences=True),
        Dropout(0.10),
        LSTM(32),
        Dropout(0.10),
        Dense(32, activation="relu"),
        Dense(FORECAST_DAYS),
    ])
    model.compile(optimizer="adam", loss="mse")
    early_stop = EarlyStopping(monitor="val_loss", patience=5, restore_best_weights=True)
    model.fit(X[:split], y[:split], epochs=40, batch_size=32,
              validation_split=0.10, callbacks=[early_stop], verbose=0)
    return model, scaler, scaled


if "started" not in st.session_state:
    st.session_state.started = False

if not st.session_state.started:
    st.markdown(
        """
        <div style="text-align:center; padding:80px 20px 40px 20px;">
            <div style="font-size:75px; margin-bottom:10px;">⚡</div>
            <h1 style="font-size:60px; font-weight:800; margin:0;">WATTWISE</h1>
            <h2 style="font-size:28px; font-weight:500; margin-top:15px;">
                Smart Electricity Consumption & Bill Forecasting
            </h2>
            <p style="font-size:18px; margin-top:25px;">
                Predict electricity consumption, estimate monthly bills,
                and visualize future energy usage using AI-powered forecasting.
            </p>
        </div>
        """,
        unsafe_allow_html=True,
    )
    _, mid, _ = st.columns([1, 2, 1])
    with mid:
        if st.button("🚀 Start Forecasting", type="primary", use_container_width=True):
            st.session_state.started = True
            st.rerun()
    st.stop()


st.title("⚡ WattWise")
st.write("Enter your electricity details to generate the forecast and estimated monthly bill.")

st.sidebar.header("Electricity Input")

consumer_type = st.sidebar.selectbox("Consumer Type", ["Household", "Industrial"])

month_names = [
    "January", "February", "March", "April", "May", "June",
    "July", "August", "September", "October", "November", "December",
]
selected_month = st.sidebar.selectbox("Forecast Month", month_names, index=1)

unit_cost = st.sidebar.number_input(
    "Electricity Unit Cost (₹/kWh)",
    min_value=1.0, max_value=100.0, value=8.0, step=0.5,
)

if consumer_type == "Household":
    number_of_people = st.sidebar.number_input(
        "Number of People", min_value=1, max_value=20, value=4, step=1
    )
    daily_consumption = st.sidebar.number_input(
        "Average Daily Consumption (kWh)",
        min_value=5.0, max_value=100.0, value=8.0, step=1.0,
    )
else:
    industry_size = st.sidebar.selectbox(
        "Industry Size", ["Small Industry", "Medium Industry", "Large Industry"]
    )
    limits = {
        "Small Industry": (50, 1000, 250),
        "Medium Industry": (500, 5000, 1000),
        "Large Industry": (2000, 20000, 5000),
    }
    minimum, maximum, default = limits[industry_size]
    daily_consumption = st.sidebar.number_input(
        "Average Daily Consumption (kWh)",
        min_value=float(minimum), max_value=float(maximum),
        value=float(default), step=50.0,
    )

st.divider()

st.subheader("Your Electricity Details")
a, b, c, d = st.columns(4)
a.metric("Consumer", consumer_type)
b.metric("Forecast Month", selected_month)
c.metric("Daily Consumption", f"{daily_consumption:.1f} kWh")
d.metric("Unit Cost", f"₹{unit_cost:.2f}")
st.divider()

if st.button("🚀 Generate Electricity Forecast", type="primary", use_container_width=True):
    status = st.empty()
    bar = st.progress(0)
    status.info("Processing... Training prediction model.")
    bar.progress(20)
    model, scaler, scaled = train_model()
    bar.progress(55)
    status.info("Processing... Generating consumption prediction.")

    LOOKBACK = 30
    seq = scaled[-LOOKBACK:].reshape(1, LOOKBACK, 1)
    preds = scaler.inverse_transform(
        model.predict(seq, verbose=0)[0].reshape(-1, 1)
    ).flatten()

    # Rescale the model output to the consumption level the user entered
    model_average = preds.mean()
    if model_average <= 0:
        model_average = 1
    preds = preds / model_average * daily_consumption

    # Keep predictions inside a realistic band around the user's average
    if consumer_type == "Household":
        lower, upper = daily_consumption * 0.75, daily_consumption * 1.25
    else:
        lower, upper = daily_consumption * 0.85, daily_consumption * 1.15
    preds = np.clip(preds, lower, upper)

    # Build the real calendar days of the selected month
    month_number = month_names.index(selected_month) + 1
    month_start = pd.Timestamp(year=2026, month=month_number, day=1)
    dates = pd.date_range(month_start, month_start + pd.offsets.MonthEnd(0), freq="D")
    n_days = len(dates)

    # Stretch the 30 day forecast to 28, 29, 30 or 31 days
    preds = np.interp(np.linspace(0, 29, n_days), np.arange(30), preds)
    bar.progress(85)
    status.info("Processing... Calculating electricity bill.")

    daily_bill = preds * unit_cost
    total_consumption = preds.sum()
    total_bill = daily_bill.sum()
    average_consumption = preds.mean()

    forecast_df = pd.DataFrame({
        "Date": dates,
        "Predicted Consumption (kWh)": np.round(preds, 2),
        "Unit Cost (₹/kWh)": unit_cost,
        "Estimated Daily Bill (₹)": np.round(daily_bill, 2),
    })

    bar.progress(100)
    status.success("Forecast generated successfully!")

    st.divider()
    st.subheader(f"{selected_month} 2026 Electricity Forecast")
    m1, m2, m3 = st.columns(3)
    m1.metric("Total Consumption", f"{total_consumption:,.2f} kWh")
    m2.metric("Average Daily Consumption", f"{average_consumption:,.2f} kWh")
    m3.metric("Estimated Monthly Bill", f"₹{total_bill:,.2f}")

    st.divider()
    st.subheader("Daily Forecast")
    st.dataframe(forecast_df, use_container_width=True, hide_index=True)

    st.divider()
    st.subheader("Daily Electricity Consumption Forecast")
    st.line_chart(forecast_df.set_index("Date")["Predicted Consumption (kWh)"])

    st.subheader("Daily Electricity Bill")
    st.line_chart(forecast_df.set_index("Date")["Estimated Daily Bill (₹)"])

    st.divider()
    st.subheader("Monthly Bill Summary")
    st.write(f"Total Predicted Consumption: {total_consumption:,.2f} kWh")
    st.write(f"Fixed Electricity Price: ₹{unit_cost:.2f} / kWh")
    st.write(f"Estimated Monthly Bill: ₹{total_bill:,.2f}")
    st.success(
        f"Your estimated electricity bill for {selected_month} 2026 "
        f"is ₹{total_bill:,.2f}"
    )
'''


with open("/content/app.py", "w", encoding="utf-8") as f:
